cmake_minimum_required(VERSION 3.24)
project(dag_execution LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Threads REQUIRED)
add_library(dag INTERFACE)
target_include_directories(dag INTERFACE src)
target_link_libraries(dag INTERFACE Threads::Threads)
option(DAG_ASAN "Address sanitizer" OFF)
option(DAG_TSAN "Thread sanitizer (Linux only)" OFF)
option(DAG_GOOGLE_BENCHMARK "Fetch pinned Google Benchmark and build microbenchmarks" OFF)
if(DAG_ASAN AND DAG_TSAN)
  message(FATAL_ERROR "Sanitizers require separate builds")
endif()
if(DAG_ASAN)
  target_compile_options(dag INTERFACE -fsanitize=address -g)
  target_link_options(dag INTERFACE -fsanitize=address)
endif()
if(DAG_TSAN)
  if(WIN32)
    message(FATAL_ERROR "TSan requires Linux")
  endif()
  target_compile_options(dag INTERFACE -fsanitize=thread -g -fno-omit-frame-pointer)
  target_link_options(dag INTERFACE -fsanitize=thread)
endif()
if(MSVC)
  target_compile_options(dag INTERFACE /W4)
else()
  target_compile_options(dag INTERFACE -Wall -Wextra -Wpedantic)
endif()
add_executable(checks tests/checks.cpp)
target_link_libraries(checks PRIVATE dag)
add_executable(solution bench/main.cpp)
target_link_libraries(solution PRIVATE dag)
enable_testing()
add_test(NAME invariants COMMAND checks)
set_tests_properties(invariants PROPERTIES TIMEOUT 120)
add_test(NAME million_nodes COMMAND checks --stress)
set_tests_properties(million_nodes PROPERTIES TIMEOUT 300 LABELS stress)
if(DAG_ASAN AND WIN32)
  execute_process(COMMAND ${CMAKE_CXX_COMPILER} -print-resource-dir OUTPUT_VARIABLE DAG_RESOURCE_DIR OUTPUT_STRIP_TRAILING_WHITESPACE)
  set(DAG_RUNTIME_DIR "${DAG_RESOURCE_DIR}/lib/windows")
  set_tests_properties(invariants million_nodes PROPERTIES ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${DAG_RUNTIME_DIR}")
endif()
if(DAG_GOOGLE_BENCHMARK)
  include(FetchContent)
  set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(benchmark GIT_REPOSITORY https://github.com/google/benchmark.git GIT_TAG v1.9.4 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(benchmark)
  add_executable(micro bench/micro.cpp)
  target_link_libraries(micro PRIVATE dag benchmark::benchmark_main)
endif()
